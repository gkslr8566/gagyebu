        return '<div class="hw-inv-c"><div class="hw-inv-n">' + esc_(short) + '</div><div class="hw-inv-v">' + man_(a.value) + '</div>' +
          (r !== null ? '<div class="' + (r >= 0 ? 'stock-pos' : 'stock-neg') + '">' + (r >= 0 ? '+' : '') + pct1_(r) + '</div>' : '<div class="hw-foot">' + esc_(a.month || '') + ' 수동 기록</div>') + '</div>';
      }).join('') + '</div><div class="hw-foot">보유수량 × 현재가 · 예수금·IRP 펀드 제외 · 수익률은 매입금액 대비</div>';
    },
    realized: function (ctx) {
      var rb = ctx.invest.realizedByMonth;
      var ms = Object.keys(rb).sort();
      if (!ms.length) return '<div class="hw-empty">매도 기록이 없어요</div>';
      var cum = 0, vals = ms.map(function (m) { cum += rb[m]; return cum; });
      var last = rb[ms[ms.length - 1]];
      return lineSvg_(ms.map(monthLabel_), vals, '#10b981', man_) +
        '<div class="hw-kv"><span>' + monthLabel_(ms[ms.length - 1]) + ' 실현손익</span><b class="' + (last >= 0 ? 'stock-pos' : 'stock-neg') + '">' + signed_(last, man_) + '</b></div>';
    },
    car: function () {
      var v = computeVehicleMonthlyData_();
      var ms = v.months.filter(function (m) { return v.mileageMap[m] > 0; }).slice(-6);
      if (!ms.length) return '<div class="hw-empty">주행거리 기록이 있는 달이 없어요</div>';
      var vals = ms.map(function (m) { var b = v.byMonth[m] || {}; return (b.충전 || 0) / v.mileageMap[m]; });
      var max = Math.max.apply(null, vals);
      return ms.map(function (m, i) {
        return '<div class="hw-kv"><span>' + monthLabel_(m) + ' · ' + Math.round(v.mileageMap[m]).toLocaleString('ko-KR') + 'km</span><b>' + Math.round(vals[i]) + '원/km</b></div>' + bar_(vals[i], max, '#a855f7', 8);
      }).join('') + '<div class="hw-foot">충전비 ÷ 주행거리 (톨비 제외)</div>';
    }
  };

  function homeDefaultMonth_(months) {
    var d = new Date();
    var cur = d.getFullYear() + '.' + ((d.getMonth() + 1) < 10 ? '0' : '') + (d.getMonth() + 1);
    var prev = prevMonthOf(cur);
    if (months.indexOf(prev) >= 0) return prev;
    return months[months.length - 1] || null;
  }
