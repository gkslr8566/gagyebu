        return res.text().then(function (text) {

          if (!data || !('rows' in data)) throw new Error('예상치 못한 응답 형식입니다: ' + text.slice(0, 120));

        });
      });
    }).then(function (result) {
      cacheSet_('cache_ledger', result.rows);
      return result;
    }).catch(function (err) {
      console.error('불러오기 실패', err);
      var cached = cacheGet_('cache_ledger');
      if (cached) return { rows: cached, error: (err && err.message) || String(err), fromCache: true };
      return { rows: null, error: (err && err.message) || String(err) };
    });

  function fetchSheetList_(sheetName) {
    var url = getServerUrl();
    if (!url) return Promise.resolve({ rows: null });
    var sep = url.indexOf('?') >= 0 ? '&' : '?';
    var fullUrl = url + sep + 'action=list&sheet=' + encodeURIComponent(sheetName);
    var cacheKey = 'cache_sheet_' + sheetName;
    return withRetry_(function () {
      return fetch(fullUrl).then(function (res) {
        return res.text().then(function (text) {
          var data;
          try { data = JSON.parse(text); }
          catch (e) { throw new Error('응답이 JSON이 아닙니다 (HTTP ' + res.status + ')'); }
          if (data && data.error) throw new Error(data.error);
          if (!data || !('rows' in data)) throw new Error('예상치 못한 응답 형식입니다.');
          return { rows: data.rows };
        });
      });
    }).then(function (result) {
      cacheSet_(cacheKey, result.rows);
      return result;
    }).catch(function (err) {
      console.error('불러오기 실패', err);
      var cached = cacheGet_(cacheKey);
      if (cached) return { rows: cached, error: (err && err.message) || String(err), fromCache: true };
      return { rows: null, error: (err && err.message) || String(err) };
    });
  }

  function fetchCurrentPrices_() {
    var url = getServerUrl();
    if (!url) return Promise.resolve({ prices: null });
    var sep = url.indexOf('?') >= 0 ? '&' : '?';
    var fullUrl = url + sep + 'action=currentPrices';
    return withRetry_(function () {
      return fetch(fullUrl).then(function (res) {
        return res.text().then(function (text) {
          var data;
          try { data = JSON.parse(text); }
          catch (e) { throw new Error('응답이 JSON이 아닙니다 (HTTP ' + res.status + ')'); }
          if (data && data.error) throw new Error(data.error);
          if (!data || !('prices' in data)) throw new Error('예상치 못한 응답 형식입니다. (Code.gs가 최신 버전으로 재배포됐는지 확인하세요)');
          return { prices: data.prices };
        });
      });
    }).then(function (result) {
      cacheSet_('cache_prices', result.prices);
      return result;
    }).catch(function (err) {
      console.error('현재가 조회 실패', err);
      var cached = cacheGet_('cache_prices');
      if (cached) return { prices: cached, error: (err && err.message) || String(err), fromCache: true };
      return { prices: null, error: (err && err.message) || String(err) };
    });
  }

  function refreshTrades() {
    var url = getServerUrl();
    if (!url) {
      state.trades.rows = loadSeedTrades();
      state.trades.prices = loadSeedPrices();
      state.trades.priceStatus = { ok: false, offline: true };
      renderTradesTab();
      renderAssetsTab();
      return;
    }
    Promise.all([fetchSheetList_(TRADES_SHEET), fetchCurrentPrices_()]).then(function (results) {
      var tradeResult = results[0];
      var priceResult = results[1];
      state.trades.rows = (tradeResult.rows === null) ? loadSeedTrades() : tradeResult.rows;
      if (priceResult.prices === null) {
        state.trades.prices = loadSeedPrices();
        state.trades.priceStatus = { ok: false, error: priceResult.error };
      } else {
        state.trades.prices = priceResult.prices;
        state.trades.priceStatus = { ok: true, count: priceResult.prices.length, fromCache: !!priceResult.fromCache };
      }
      renderTradesTab();
      renderAssetsTab();
    });
  }

  function parseTradeRow(row) {
    return {
      id: row['ID'],
      yearMonth: normStr(row['연월']),
      date: row['구매일'],
      account: normStr(row['증권사']),
      action: normStr(row['매매구분']),
      name: normStr(row['종목명']),
      code: normStr(row['종목코드']),
      market: row['시장구분'] || '',
      sector: row['섹터'] || '',
      price: parseAmount(row['평단가']),
      qtyInput: parseAmount(row['입력수량']),
      qtySigned: parseAmount(row['계산수량']),
      rate: parseAmount(row['환율']) || 1,
      note: row['비고'] || ''
    };
  }

  /**
   * 매매 원본 데이터를 계좌+종목명 기준으로 다시 계산한다.
   * (구글시트의 기존 실평단가 수식은 종목코드만으로 누적 평균을 구해서
   *  같은 종목을 여러 계좌에서 매매하면 실현손익이 섞이는 버그가 있었음.
   *  Code.gs의 recomputeTradeCalculations_()와 동일한 로직으로, 시트가
   *  아직 고쳐지지 않았어도 화면에는 항상 정확한 값이 보이도록 한다.)
   */
  function computeTradeState(rawRows) {
    var rows = rawRows.map(parseTradeRow).filter(function (r) { return r.id; });
    var indexed = rows.map(function (r, i) { return { row: r, idx: i }; });
    indexed.sort(function (a, b) {
      var da = a.row.date || '', db = b.row.date || '';
      if (da !== db) return da < db ? -1 : 1;
      return a.idx - b.idx;
    });

    var holdings = {}; // "계좌||종목명" -> { account, name, code, sector, qty, cost, avgDateMs }

    indexed.forEach(function (item) {
      var r = item.row;
      var key = r.account + '||' + r.name;
      if (!holdings[key]) holdings[key] = { account: r.account, name: r.name, code: r.code, sector: r.sector, qty: 0, cost: 0, avgDateMs: null };
      var st = holdings[key];
      var tradeMs = r.date ? new Date(r.date).getTime() : null;

      var avgCost, realizedPnl;
      if (r.action === '매도') {
        avgCost = st.qty > 0 ? st.cost / st.qty : 0;
