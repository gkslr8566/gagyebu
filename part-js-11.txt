      if (codeEl && info.code && (!codeEl.value || codeEl.value === codeEl.dataset.autofilled)) {
        codeEl.value = info.code;
        codeEl.dataset.autofilled = info.code;
      }
      if (sectorEl && info.sector && (!sectorEl.value || sectorEl.value === sectorEl.dataset.autofilled)) {
        sectorEl.value = info.sector;
        sectorEl.dataset.autofilled = info.sector;
