        '<td><div class="mini-bar-cell"><div class="mini-bar-track"><div class="mini-bar-fill" style="width:' + (restRatio * 100) + '%; background:var(--muted);"></div></div><span>' + pct(restRatio) + '</span></div></td>' +
        '<td style="text-align:right;">-</td>' +
        '</tr>';
    }

    tbody.innerHTML = rowsHtml;
  }

  function renderTable(rows) {
    var sorted = rows.slice().sort(function (a, b) { return (b.date || '').localeCompare(a.date || ''); });
    var visible = sorted.slice(0, state.visibleCount);
    var tbody = document.getElementById('txTableBody');
    var empty = document.getElementById('txEmpty');

    if (!visible.length) {
      tbody.innerHTML = '';
      empty.style.display = 'block';
    } else {
      empty.style.display = 'none';
      tbody.innerHTML = visible.map(function (r) {
        var amtClass = r.type === '수입' ? 'amt-pos' : 'amt-neg';
        var sign = r.type === '수입' ? '+' : '-';
        return '<tr>' +
          '<td>' + (r.date || '') + '</td>' +
          '<td><span class="tag">' + r.type + '</span></td>' +
          '<td>' + (r.main || '') + '</td>' +
          '<td>' + (r.sub || '') + '</td>' +
          '<td>' + (r.memo || '') + '</td>' +
          '<td>' + (r.user || '') + '</td>' +
          '<td>' + (r.payment || '') + '</td>' +
          '<td style="text-align:right;" class="' + amtClass + '">' + sign + won(Math.abs(r.applied)).replace('₩','₩') + '</td>' +
          '</tr>';
      }).join('');
    }

    document.getElementById('loadMoreBtn').style.display = sorted.length > state.visibleCount ? 'inline-block' : 'none';
  }

  // ---------- 입력 폼 ----------
  function updateCategoryOptions() {
    var type = document.getElementById('typeSelect').value;
    var meta = window.__categoryList || [];
    var filtered = meta.filter(function (c) { return c.indexOf(type + ' >') === 0 || c.indexOf(type + '>') === 0; });
    var sel = document.getElementById('categorySelect');
    if (!filtered.length) {
      // 메타 정보가 없으면 현재 데이터에서 유추
      var set = new Set();
      state.rows.forEach(function (r) {
        if (r['수입/지출'] === type && r['분류']) set.add(r['분류']);
      });
      filtered = Array.from(set);
    }
    sel.innerHTML = filtered.map(function (c) { return '<option value="' + c + '">' + c + '</option>'; }).join('');
  }

  function fetchMetaAndFillCategories() {
    var url = getServerUrl();
    if (!url) { window.__categoryList = []; updateCategoryOptions(); return; }
    var sep = url.indexOf('?') >= 0 ? '&' : '?';
    fetch(url + sep + 'action=meta').then(function (r) { return r.json(); }).then(function (data) {
      window.__categoryList = data.categoryList || [];
      updateCategoryOptions();
    }).catch(function () {
      window.__categoryList = [];
      updateCategoryOptions();
    });
  }

  function submitEntry(ev) {
    ev.preventDefault();
    var form = document.getElementById('entryForm');
    var statusEl = document.getElementById('formStatus');
    var url = getServerUrl();

    var entry = {
      date: form.date.value,
      type: form.type.value,
      category: form.category.value,
      amount: Number(form.amount.value) || 0,
      discount: Number(form.discount.value) || 0,
      user: form.user.value,
      paymentMethod: form.paymentMethod.value,
      memo: form.memo.value,
      note: form.note.value
    };

    if (!url) {
      statusEl.className = 'status-text status-err';
      statusEl.textContent = '⚠ 시트 연동이 설정되지 않아 저장할 수 없습니다. 상단 "시트 연동 설정"에서 URL을 등록하세요.';
      return;
    }

    statusEl.className = 'status-text';
    statusEl.textContent = '저장 중...';
    document.getElementById('submitEntryBtn').disabled = true;

    fetch(url, {
      method: 'POST',
      body: JSON.stringify({ action: 'addEntry', entry: entry })
    }).then(function (res) { return res.json(); })
      .then(function (data) {
        document.getElementById('submitEntryBtn').disabled = false;
        if (data.ok) {
          statusEl.className = 'status-text status-ok';
          statusEl.textContent = '✓ 저장 완료! 목록을 새로고침합니다.';
          form.reset();
          refreshAll();
        } else {
          statusEl.className = 'status-text status-err';
          statusEl.textContent = '저장 실패: ' + (data.error || '알 수 없는 오류');
        }
      }).catch(function (err) {
        document.getElementById('submitEntryBtn').disabled = false;
        statusEl.className = 'status-text status-err';
        statusEl.textContent = '저장 실패: ' + err;
      });
  }
