  function fetch(url, opts) {
    var key = getAppKey_();
    var u = String(url);
    if (opts && opts.method === 'POST' && typeof opts.body === 'string') {
      try {
        var p = JSON.parse(opts.body);
        p.key = key;
        opts = Object.assign({}, opts, { body: JSON.stringify(p) });
      } catch (e) { /* JSON이 아니면 그대로 보낸다 */ }
    } else if (key) {
      u += (u.indexOf('?') >= 0 ? '&' : '?') + 'key=' + encodeURIComponent(key);
    }
    return nativeFetch_(u, opts).then(function (res) {
      try {
        res.clone().json().then(function (d) { if (d && d.error === 'unauthorized') onUnauthorized_(); }).catch(function () {});
      } catch (e) { /* 응답을 복제할 수 없으면 건너뛴다 */ }
      return res;
    });
  }

  function makeValueLabelPlugin_(formatter) {
    return {
      id: 'valueLabelPlugin',
      afterDatasetsDraw: function (chart) {
        var ctx = chart.ctx;
        chart.data.datasets.forEach(function (dataset, dsIndex) {
          var meta = chart.getDatasetMeta(dsIndex);
          if (!meta || meta.hidden) return;
          meta.data.forEach(function (el, i) {
            var raw = dataset.data[i];
            if (raw === null || raw === undefined) return;
            var text = formatter(raw);
            var isBar = typeof el.base === 'number';
            var y = isBar ? (raw >= 0 ? el.y - 6 : el.y + 14) : el.y - 10;
            ctx.save();
            ctx.fillStyle = '#1f2430';
            ctx.font = "600 10px -apple-system, BlinkMacSystemFont, 'Apple SD Gothic Neo', sans-serif";
            ctx.textAlign = 'center';
            ctx.fillText(text, el.x, y);
            ctx.restore();
          });
        });
      }
    };
  }

      emptyEl.textContent = fallbackMessage || ('⚠ 차트를 그리지 못했어요: ' + ((e && e.message) || e));

  function parseAmount(v) {
    if (v === undefined || v === null || v === '') return 0;
    if (typeof v === 'number') return v;
    var cleaned = String(v).replace(/[₩,\s]/g, '');
    var n = parseFloat(cleaned);
    return isNaN(n) ? 0 : n;

  function prevMonthOf(m) {
    if (!m) return null;
    var parts = m.split('.');
    var y = parseInt(parts[0], 10), mo = parseInt(parts[1], 10);
    if (isNaN(y) || isNaN(mo)) return null;
    mo -= 1;
    if (mo === 0) { mo = 12; y -= 1; }
    return y + '.' + (mo < 10 ? '0' + mo : '' + mo);
  }

  function delay_(ms) {

  function withRetry_(taskFn, attempts) {
    attempts = attempts || 3;
    function attempt(n) {
      return taskFn().catch(function (err) {
        if (n >= attempts) throw err;
        return delay_(500 * n).then(function () { return attempt(n + 1); });
      });
    }
    return attempt(1);
  }
